cmake_minimum_required(VERSION 3.16)
project(GeometryOnTheGPU LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
foreach(config RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} ${CMAKE_BINARY_DIR})
endforeach()
find_package(Threads REQUIRED)
add_library(lab STATIC src/core/trace_field.cpp src/core/mesh.cpp src/core/image.cpp src/core/metric.cpp src/core/simplify.cpp src/core/judge_metric.cpp
  src/cpu/part0_hello.cpp src/cpu/part1_sdf.cpp src/cpu/part2_qem.cpp src/cpu/part3_metric.cpp)
target_link_libraries(lab PUBLIC Threads::Threads)
foreach(program hello trace simplify metric judge check)
  add_executable(${program} src/${program}_main.cpp)
  target_link_libraries(${program} PRIVATE lab)
endforeach()
add_executable(make_part tools/make_part.cpp)
target_link_libraries(make_part PRIVATE lab)
enable_testing()
add_test(NAME student_tasks COMMAND check --ci)
set_tests_properties(student_tasks PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})

# Instructor verification of the given machinery.
add_executable(verify_given tools/verify_given.cpp)
target_link_libraries(verify_given PRIVATE lab)
add_test(NAME given_helpers COMMAND verify_given)
